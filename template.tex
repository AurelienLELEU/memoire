% --- Numérotation hiérarchique ---
% Les Parties (#) restent des sections non numérotées (séparateurs).
% Les Chapitres (##) deviennent l'unité numérotée de plus haut niveau (1, 2, 3, ...).
% Les sections (###) et sous-sections (####) héritent : 4.1, 4.1.2, etc.
\setcounter{secnumdepth}{4}
\renewcommand{\thesubsection}{\arabic{subsection}}

% --- Pages remplies jusqu'en bas ---
% Par défaut, la classe « article » utilise \raggedbottom : chaque page
% se termine où le texte se termine naturellement, ce qui peut laisser
% beaucoup d'espace blanc en bas, surtout quand la page suivante commence
% par un titre de section (LaTeX préfère pousser le titre à la page
% suivante plutôt que de le laisser seul en bas).
% \flushbottom force LaTeX à étirer les espaces verticaux élastiques
% (autour des titres, entre paragraphes) pour aligner toutes les pages
% sur la même hauteur. Les pénalités widow/club sont aussi relâchées
% pour donner plus de souplesse au moteur de coupure.
\flushbottom
\widowpenalty=1000
\clubpenalty=1000

% --- Titres : forcer \paragraph et \subparagraph à passer à la ligne ---
% Par défaut LaTeX rend \paragraph (= #### Markdown) en titre « run-in »,
% collé au début du paragraphe qui suit. On les redéfinit en blocs.
\usepackage{titlesec}
\titleformat{\paragraph}[block]
  {\normalfont\normalsize\bfseries}{\theparagraph}{1em}{}
\titlespacing*{\paragraph}
  {0pt}{2.25ex plus 1ex minus .2ex}{1ex plus .2ex}
\titleformat{\subparagraph}[block]
  {\normalfont\normalsize\bfseries\itshape}{\thesubparagraph}{1em}{}
\titlespacing*{\subparagraph}
  {0pt}{2ex plus 1ex minus .2ex}{0.8ex plus .2ex}

% Chaque Chapitre (= \subsection) commence sur une nouvelle page, avec un
% libellé « Chapitre N — » prépendu automatiquement à son intitulé. La page
% d'ouverture de chapitre utilise le style « plain » (pas d'en-tête).
\titleformat{\subsection}[block]
  {\clearpage\thispagestyle{plain}\normalfont\LARGE\bfseries}
  {Chapitre~\thesubsection~\textemdash}{0.6em}{}
\titlespacing*{\subsection}
  {0pt}{0pt}{1.5ex plus .3ex}

% --- Anti-débordement (inline code, longs identifiants, URLs) ---
% Pandoc convertit `code` en \texttt{...}. Par défaut, la fonte monospace
% interdit toute coupure : les identifiants type « recursive-1024-128 »
% débordent en marge.
% Stratégie :
%   (a) marge d'élasticité d'urgence + \sloppy,
%   (b) coupure aux tirets « - » réactivée dans la fonte monospace,
%   (c) le paquet « underscore » rend « \_ » sécable en mode texte,
%       ce qui permet le retour à la ligne après un « _ » dans les
%       identifiants type « entité_émettrice », sans couper le mot
%       au milieu d'une syllabe.
\setlength{\emergencystretch}{4em}
\usepackage[strings]{underscore}

\makeatother
